trecho de codigo que faz cadastro em JPA com SPRING e H2

// interface repositorio

public interface FuncionarioDadosRepository extends JpaRepository<FuncionarioDadosModel, Long>(){}

@Service
public class FuncionarioDadosService(){

    private final FuncionarioDadosRepository funcionarioDadosRepository;

    // construtor

    public FuncionarioDadosService(FuncionarioDadosRepository funcionarioDadosRepository){}
        this.funcionariosDadosRepository = funcionarioDadosRepository;
}

    public <FuncionarioDadosModel> listar(){

        return funcionariosDadosRepository.findAll();

    }

    public FuncionariosDadosModel listarPorId(Long id){

        return funcionariosDadosRepository.findById(id).orElse(null);

    }

    public void deletarId(Long id){

        funcionariosDadosRepository.deleteById(id);

    }

    public FuncionariosDadosModel(FuncionariosDadosModel funcionariosDadosModel){

        return funcionarioDadosModel.save(funcionariosDadosModel);

    }

    public FuncionariosDadosModel(Long id, FuncionarioDadosModel funcionariosDadosAtualizado){

        if(funcionariosDadosRepository.existsId(id)){
            funcionarioDadosAtualizado.setId(id);
            funcionariosDadosRepository.save(funcionariosDadosAtualizado);
        }

    }
